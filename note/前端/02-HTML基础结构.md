<html lang="en">

        <meta charset="UTF-8">
        <!-- 视口 -->
        <meta name="viewport"  content="width=device-width, initial-scale=10.0">
        <!-- 标题 -->
        <title>document</title>
    </head>
    <!-- 网页主体部分，内容区域，html结构的代码都书写在body标签重 -->
    <body>
        网页的正文内容，图片，列表，表格
    </body>
     
</html>
