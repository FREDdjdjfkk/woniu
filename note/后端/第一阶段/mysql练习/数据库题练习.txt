    c.name AS 公司名,
    COUNT(CASE WHEN d.name = '市场部' THEN e.id END) AS 市场部,
    COUNT(CASE WHEN d.name = '教职部' THEN e.id END) AS 教职部,
    COUNT(CASE WHEN d.name = '教学部' THEN e.id END) AS 教学部,
    COUNT(CASE WHEN d.name = '财务部' THEN e.id END) AS 财务部

LEFT JOIN department d ON d.companyId = c.id
LEFT JOIN employee e ON e.deptId = d.id
WHERE c.name LIKE '成都蜗牛%'
GROUP BY c.name;

31）查询公司规模在100-200人之间的所有员工

SELECT
*
FROM employee e
LEFT JOIN department d ON e.deptId = d.id
LEFT JOIN company c ON d.companyId = c.id
WHERE c.id IN (
    SELECT c.id
    FROM company c
    JOIN department d ON d.companyId = c.id
    GROUP BY c.id
    HAVING SUM(d.number) BETWEEN 100 AND 200
)

41）查询所有员工中，入职5年以内和入职5年以上的员工对应人数

SELECT 
    CASE 
        WHEN TIMESTAMPDIFF(YEAR, e.joinDate, CURDATE()) < 5 THEN '入职5年以内'
        ELSE '入职5年以上'
    END AS 入职年限分类,
		COUNT(e.id)
FROM employee e
LEFT JOIN department d ON e.deptId = d.id
LEFT JOIN company c ON d.companyId = c.id
WHERE c.name LIKE '成都蜗牛学苑%'
GROUP BY 入职年限分类;

42）查询每个公司每个月的总支出薪水，并按照从低到高排序

SELECT
c.`name` 公司,
SUM(e.salary) 每个月总薪水
FROM employee e
LEFT JOIN department d ON e.deptId = d.id
LEFT JOIN company c ON d.companyId = c.id
GROUP BY c.`name`
ORDER BY SUM(e.salary) DESC;
